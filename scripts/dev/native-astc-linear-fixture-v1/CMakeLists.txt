cmake_minimum_required(VERSION 3.20)
project(dm_native_astc_linear_matrix C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(PRODUCER "${ROOT}/scripts/dev/native-image-payload")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_program(GLSLANG glslangValidator REQUIRED)
find_program(SPIRV_VAL spirv-val REQUIRED)
set(SPV "${CMAKE_CURRENT_BINARY_DIR}/native_sample_spv.h")
add_custom_command(OUTPUT "${SPV}"
 COMMAND "${Python3_EXECUTABLE}" "${ROOT}/dm/gfx/vulkan/build_builtin.py" --source "${PRODUCER}/sample.comp" --output "${SPV}" --symbol dm_native_sample_spv --glslang "${GLSLANG}" --validator "${SPIRV_VAL}"
 DEPENDS "${PRODUCER}/sample.comp" "${ROOT}/dm/gfx/vulkan/build_builtin.py" VERBATIM)
set(objects "")
set(matrix_widths 1 5 63 64 95 128 128 256)
set(matrix_heights 1 7 65 64 97 128 256 256)
set(matrix_levels 1 3 7 7 7 8 9 9)
set(source_identity "")
foreach(name fixture.c native_import.inc resource_bake.inc resource_batch.inc resource_format.h sample.comp native_linear_roundtrip.inc)
 file(SHA256 "${PRODUCER}/${name}" digest)
 string(APPEND source_identity "${name}:${digest}\n")
endforeach()
file(SHA256 "${CMAKE_CURRENT_LIST_FILE}" cmake_sha)
string(APPEND source_identity "matrix-CMakeLists.txt:${cmake_sha}\ncontract-profile:turnip-coherent-host-astc-v1\n")
string(SHA256 fingerprint "${source_identity}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/producer-source-sha256.txt" "${fingerprint}\n")
foreach(index RANGE 0 7)
 list(GET matrix_widths ${index} width)
 list(GET matrix_heights ${index} height)
 list(GET matrix_levels ${index} levels)
 add_library(shape_${index} OBJECT "${PRODUCER}/fixture.c" "${SPV}")
 target_include_directories(shape_${index} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${ROOT}/dm/third_party/volk" "${ROOT}/dm/third_party/runtime-codecs")
 target_compile_definitions(shape_${index} PRIVATE DM_NATIVE_IMPORTED_IMAGE=1 DM_NATIVE_TURNIP_HOST_ASTC=1 DM_NATIVE_IMAGE_EMBED=1 DM_NATIVE_LINEAR_MATRIX=1 DM_NATIVE_MATRIX_WIDTH=${width} DM_NATIVE_MATRIX_HEIGHT=${height} DM_NATIVE_MATRIX_LEVELS=${levels} DM_NATIVE_FIXTURE_ENTRY=dm_native_linear_shape_${index} MBEDTLS_CONFIG_FILE="dm_sha_config.h" DM_NATIVE_RESOURCE_PRODUCER_SHA="${fingerprint}")
 target_compile_options(shape_${index} PRIVATE -O2 -g -Wall -Wextra -Werror -fPIC)
 list(APPEND objects "$<TARGET_OBJECTS:shape_${index}>")
endforeach()
add_library(native_astc_linear_matrix SHARED matrix_jni.c ${objects} "${ROOT}/dm/third_party/volk/volk.c" "${ROOT}/dm/third_party/runtime-codecs/sha256.c")
target_include_directories(native_astc_linear_matrix PRIVATE "${ROOT}/dm/third_party/volk" "${ROOT}/dm/third_party/runtime-codecs" "${ROOT}/dm/third_party/libadrenotools/include")
target_compile_options(native_astc_linear_matrix PRIVATE -O2 -g -Wall -Wextra -Werror)
target_compile_definitions(native_astc_linear_matrix PRIVATE MBEDTLS_CONFIG_FILE="dm_sha_config.h")
target_link_options(native_astc_linear_matrix PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(native_astc_linear_matrix PRIVATE dl m)
