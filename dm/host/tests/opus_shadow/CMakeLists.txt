cmake_minimum_required(VERSION 3.20)
project(opus_shadow_test C)
enable_testing()
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../third_party/opus_fixed.cmake")
dm_add_fixed_opus()
find_package(Threads REQUIRED)
add_executable(opus_shadow_test ../t_opus_shadow.c)
target_include_directories(opus_shadow_test PRIVATE ../..)
target_compile_definitions(opus_shadow_test PRIVATE DM_OPUS_SHADOW_OWNED_HOST=1)
target_compile_options(opus_shadow_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(opus_shadow_test PRIVATE opus Threads::Threads m)
add_test(NAME shadow_lifecycle COMMAND opus_shadow_test)
set_tests_properties(shadow_lifecycle PROPERTIES TIMEOUT 30)
